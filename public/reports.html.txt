      <h1>Attendance reports</h1>
    </div>

    <a href="index.html" style="color: white">Dashboard</a>
  </header>

  <main class="wrap">
    <section class="hero">
      <h2>Event attendance reports</h2>
      <p>
        Review attendance summaries and print or save a report as PDF.
      </p>
    </section>

    <section class="panel">
      <div class="row no-print">
        <div>
          <label for="event">Event</label>
          <select id="event" onchange="drawReports()">
            <option value="">All events</option>
          </select>
        </div>

        <div>
          <label for="query">Search student</label>
          <input
            id="query"
            type="search"
            placeholder="Enter name or student ID"
            oninput="drawReports()"
          >
        </div>

        <div style="flex: 0 0 auto">
          <button type="button" onclick="window.print()">
            Print / Save PDF
          </button>
        </div>
      </div>

      <div id="reports"></div>
    </section>
  </main>

  <script>
    requireAdmin();

    // Populate the event selector.
    system.getEvents().forEach(event => {
      $('event').insertAdjacentHTML(
        'beforeend',
        `<option value="${event.id}">${esc(event.name)}</option>`
      );
    });

    function drawReports() {
      const query = $('query').value.trim().toLowerCase();
      const selectedEventId = $('event').value;

      const events = system.getEvents().filter(event => {
        return !selectedEventId || event.id === Number(selectedEventId);
      });

      if (events.length === 0) {
        $('reports').innerHTML = `
          <p class="empty">No events to report yet.</p>
        `;
        return;
      }

      $('reports').innerHTML = events.map(event => {
        const summary = system.summary(event);

        const attendees = event.attendees.filter(student => {
          const searchableText =
            `${student.first} ${student.middle || ''} ${student.last} ${student.studentId}`
              .toLowerCase();

          return searchableText.includes(query);
        });

        return `
          <article class="panel" style="margin-top: 16px">
            <div class="title-row">
              <div>
                <h2>${esc(event.name)}</h2>
                <p>
                  ${esc(event.date)}
                  · ${fmt12(event.start)}–${fmt12(event.end)}
                  · ${esc(event.venue)}
                </p>
              </div>

              <span class="badge ${eventStatus(event)}">
                ${eventStatus(event)}
              </span>
            </div>

            <div class="stats">
              ${Object.entries({
                Registered: event.attendees.length,
                ...summary
              })
                .map(([label, count]) => `
                  <div class="stat">
                    <b>${count}</b>
                    ${label}
                  </div>
                `)
                .join('')}
            </div>

            <div class="table-wrap">
              <table>
                <thead>
                  <tr>
                    <th>#</th>
                    <th>Student ID</th>
                    <th>Name</th>
                    <th>Department</th>
                    <th>Program / section</th>
                    <th>Time in</th>
                    <th>Status</th>
                    <th>Proof</th>
                  </tr>
                </thead>

                <tbody>
                  ${
                    attendees.length
                      ? attendees.map((student, index) => `
                          <tr>
                            <td>${index + 1}</td>
                            <td>${esc(student.studentId)}</td>
                            <td>
                              ${esc(student.first)}
                              ${esc(student.middle || '')}
                              ${esc(student.last)}
                            </td>
                            <td>${esc(student.department || '')}</td>
                            <td>
                              ${esc(student.course || '')}
                              ${esc(student.section || '')}
                            </td>
                            <td>
                              ${student.timeIn ? fmt12(student.timeIn) : '—'}
                            </td>
                            <td>
                              ${system.attendeeStatus(event, student)}
                            </td>
                            <td>
                              ${student.proof ? 'Photo submitted' : '—'}
                            </td>
                          </tr>
                        `).join('')
                      : `
                        <tr>
                          <td colspan="8">No matching students.</td>
                        </tr>
                      `
                  }
                </tbody>
              </table>
            </div>
          </article>
        `;
      }).join('');
    }

    drawReports();
  </script>
